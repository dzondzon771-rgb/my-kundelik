SMART STAROSTA v10 — FIRESTORE ЕРЕЖЕЛЕРІНЕ ҚОСЫЛАТЫН БЛОК
=======================================================

МАҢЫЗДЫ: БҰЛ ТОЛЫҚ FIRESTORE RULES ЕМЕС!
Firebase Console → Firestore Database → Rules ішіндегі БҰРЫНҒЫ
ережелерді өшірмеңіз. Тек match /databases/{database}/documents { ... }
ішіне (басқа match блоктарымен қатар) төмендегі match блогын қосыңыз.

// Оқуға рұқсат: жазбаның иесі немесе сол бөлімнің меңгерушісі.
match /reports/{ownerUid}/lateArrivals/{entryId} {
  function hasDepartmentAccess() {
    let rolePath = /databases/$(database)/documents/staffRoles/$(request.auth.uid);
    return exists(rolePath)
      && get(rolePath).data.role == 'department_head'
      && get(rolePath).data.departmentId is string
      && exists(/databases/$(database)/documents/departments/$(get(rolePath).data.departmentId)/members/$(ownerUid));
  }

  function validLateEntry() {
    return request.resource.data.keys().hasOnly(['date', 'studentId', 'name', 'time'])
      && request.resource.data.keys().hasAll(['date', 'studentId', 'name', 'time'])
      && request.resource.data.date is string
      && request.resource.data.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
      && request.resource.data.studentId is string
      && request.resource.data.studentId.size() <= 120
      && request.resource.data.studentId.size() > 0
      && request.resource.data.name is string
      && request.resource.data.name.size() <= 150
      && request.resource.data.time is string
      && request.resource.data.time.matches('^([01][0-9]|2[0-3]):[0-5][0-9]$');
  }

  allow get, list: if request.auth != null &&
    (request.auth.uid == ownerUid || hasDepartmentAccess());

  allow create, update: if request.auth != null
    && request.auth.uid == ownerUid
    && validLateEntry();

  allow delete: if request.auth != null
    && request.auth.uid == ownerUid;
}

өзгертпейді. Жарияламай тұрып Firebase Rules Simulator-мен тексеріңіз.

бар болса, сол ереже бұл шектеулерді айналып өтуі мүмкін. Оларды да тексеріңіз.
